cmake_minimum_required(VERSION 3.21)
project(PavoTTSNative VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Network WebSockets Multimedia)
qt_add_executable(PavoTTSNative src/main.cpp src/ChatController.h src/ChatController.cpp)
if(WIN32)
    set_target_properties(PavoTTSNative PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
qt_policy(SET QTP0001 NEW)
qt_policy(SET QTP0002 NEW)
qt_policy(SET QTP0004 NEW)
qt_add_qml_module(PavoTTSNative URI PavoTTS VERSION 1.0 QML_FILES qml/Main.qml)
target_link_libraries(PavoTTSNative PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::WebSockets Qt6::Multimedia)
if(ANDROID)
    set_property(TARGET PavoTTSNative PROPERTY QT_ANDROID_PACKAGE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/android")
    set_property(TARGET PavoTTSNative PROPERTY QT_ANDROID_PACKAGE_NAME "com.ale5is.pavotts")
    set_property(TARGET PavoTTSNative PROPERTY QT_ANDROID_MIN_SDK_VERSION 28)
    set_property(TARGET PavoTTSNative PROPERTY QT_ANDROID_TARGET_SDK_VERSION 35)
endif()
install(TARGETS PavoTTSNative BUNDLE DESTINATION . RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
